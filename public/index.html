async function requestNextLead() { const city = document.getElementById('cityInput').value.trim(); const setterName = document.getElementById('setterName').value; if (!city) return alert('Inserisci una città!'); try { const res = await fetch('/api/next-lead', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ city, setterName }) }); const data = await res.json(); if (data.error) { alert(`Errore: ${data.error}`); return; } // Mostra la scheda del contatto estratto renderLeadCard(data.lead); } catch (err) { alert('Errore di connessione al server.'); } }
